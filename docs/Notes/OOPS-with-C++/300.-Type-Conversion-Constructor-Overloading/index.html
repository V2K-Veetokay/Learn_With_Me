#include<iostream>

using namespace std;

class Time{
	int h, m;
	public:
	    Time() { h=m=0; }
	    Time(int x, int y) { h=x; m=y; }
	    Time(const Time &t) { h =t.h; m=t.m; }
	    
	    // Conversion from int to Time
	    Time(int t) {
	        h = t/60;
	        m = t%60;
	    }
	    
	    // Conversion from Time to int
	    inline operator int() {
	        return h*60 + m;
	    }
	    
	    // Getter methods for Distance conversion
	    int getHours() const { return h; }
	    int getMinutes() const { return m; }
	    
	    void display() {
	        cout << h << "h " << m << "m" << endl;
	    }
};

class Distance {
    double km;
	public:
	    Distance() { km=0; }
	    Distance(double k) { km=k; }
	
	    // Conversion from Time to Distance (5 km/h speed)
	    Distance(const Time &t) {
	        double totalHours = t.getHours() + (t.getMinutes()/60.0);
	        km = 5.0 * totalHours;
	    }
	
	    void display() {
	        cout << km << " km" << endl;
	    }
};

int main() {
    // User-defined to built-in
    Time t1(2, 30);
    int minutes = t1;
    cout << "Minutes: " << minutes << endl;  // Output: 150
    
    // Built-in to user-defined
    Time t2 = 150;
    cout << "Time from 150 minutes: ";
    t2.display();  // Output: 2h 30m
    
    // User-defined to user-defined
    Distance d1 = t1;
    cout << "Distance traveled: ";
    d1.display();  // Output: 12.5 km
    
    return 0;
}
