class Employee{
	private:

		float salary;
		char address[20];
		
	public:
		void set_emp_id(int id){
			emp_id = id;
		}
		void set_salary(float s){
			salary = s;
		}
		void set_address(count char add[]){
			strncpy(address, add, 19);
			address[19]='\0';
		}
		int get_employee(){
			return emp_id;
		}
		float get_salary(){
			return salary
		}
		const char* get_address() const{
			return address;
		}
};

int main(){
	Employee emp1;
	emp1.set_emp_id(101);
	emp1.set_salary(50000.50);
	emp1.set_address("123 Main Street");
	
	cout<<"Employee ID: "<<emp1.get_emp_id()<<endl;
	cout<<"Salary: "<<emp1.get_salary()<<endl;

}
